const map = L.map('mapid').setView([42.87, 74.6], 5);
L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
attribution: '&copy; OpenStreetMap contributors'
}).addTo(map);

const places = [
{ name: "Алматы", coords: [43.2389, 76.8897] },
{ name: "Бишкек", coords: [42.8746, 74.5698] },
{ name: "Самарканд", coords: [39.6542, 66.9597] }
];
places.forEach(p => {
L.marker(p.coords).addTo(map).bindPopup(`<b>${p.name}</b>`);
});
